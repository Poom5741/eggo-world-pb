    | Thai Pattern | English Replacement |
    |--------------|---------------------|
    | "กรุณาอนุมัติ USDT ก่อน" | "Approve USDT first" |

    | "โปรดลองอีกครั้ง" | "Please try again" |

    ```typescript
    // In error handling
    const errorMessages: Record<string, string> = {
      'INSUFFICIENT_BALANCE': 'Insufficient balance. Please add more USDT to your wallet.',
      'APPROVAL_REQUIRED': 'Approve USDT first. Click "Approve USDT" button before purchasing.',
      'TRANSACTION_REVERTED': 'Transaction failed. Reason: {reason}. Please try again.',
      'WALLET_NOT_FOUND': 'Wallet not found. Please connect your wallet.',
      'NETWORK_ERROR': 'Network error. Please check your connection and try again.',
      'USER_REJECTED': 'Transaction rejected by user.',
    }
    
    function getErrorMessage(error: any): string {
      if (error.code === 'INSUFFICIENT_BALANCE') {
        return errorMessages.INSUFFICIENT_BALANCE
      }
      // ... handle other codes
      return error.message || 'An unexpected error occurred'
    }
    ```
    
    **Display pattern (per D-06):**
    ```typescript

      <AlertTitle>Error</AlertTitle>
      <AlertDescription>

          <Link href={`https://bscscan.com/tx/${txHash}`} className="text-sm text-blue-600">
            View transaction →
          </Link>
        )}
      </AlertDescription>
    </Alert>
    ```
