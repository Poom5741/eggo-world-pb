# Plan 36-01: NFT Burn, KYC Toggle, and spendUSDT

## Gaps: BURN-01, KYC-01, SPEND-01

**Problem:** Three remaining spec gaps:

1. **BURN-01:** No `burnNFT()` function to destroy NFTs
2. **KYC-01:** No `setKYCRequired()` toggle for withdrawal KYC enforcement
3. **SPEND-01:** No explicit `spendUSDT()` function for internal USDT spending

**Goal:** Implement all three functions in smart contract and expose via PocketBase hooks.

## Implementation

### 1. Smart Contract: Add to EggNFT.sol and AdminConfig.sol

#### BURN-01: burnNFT() in EggNFT.sol

```solidity
// In EggNFT.sol

enum NFTType { Egg, Food, Animal }

/// @notice Burn an NFT (owner or admin)
/// @param nftId NFT token ID to burn
/// @param nftType Type of NFT (Egg, Food, Animal)
function burnNFT(uint256 nftId, NFTType nftType) external {
    address owner;

    if (nftType == NFTType.Egg) {
        owner = eggNFT.ownerOf(nftId);
        require(
            owner == msg.sender || isAdmin(msg.sender),
            "Not owner or admin"
        );
        // Verify egg is not hatched (can't burn hatched eggs)
        require(!eggs[nftId].isHatched, "Cannot burn hatched egg");
        eggNFT.burn(nftId);
    } else if (nftType == NFTType.Food) {
        owner = foodNFT.ownerOf(nftId);
        require(
            owner == msg.sender || isAdmin(msg.sender),
            "Not owner or admin"
        );
        foodNFT.burn(nftId);
    } else if (nftType == NFTType.Animal) {
        owner = animalNFT.ownerOf(nftId);
        require(
            owner == msg.sender || isAdmin(msg.sender),
            "Not owner or admin"
        );
        // Verify animal is not in breeding cooldown
        require(
            animals[nftId].breedCooldownUntil < block.timestamp,
            "Animal is breeding"
        );
        animalNFT.burn(nftId);
    }

    emit NFTBurned(nftId, nftType, owner, msg.sender);
}

event NFTBurned(uint256 nftId, NFTType nftType, address owner, address burner);
```

```solidity
bool public kycRequired = false; // Default: KYC not required

/// @notice Toggle KYC requirement for withdrawals
/// @param required Whether KYC is required
function setKYCRequired(bool required) external onlyOwner {
    kycRequired = required;
    emit KYCRequirementUpdated(required);
}

/// @notice Check if user is KYC verified
function isKYCVerified(address user) public view returns (bool) {
    return kycVerified[user];
}

/// @notice Mark user as KYC verified (admin only)
function markKYCVerified(address user, bool verified) external onlyOwner {
    kycVerified[user] = verified;
    emit KYCStatusUpdated(user, verified);
}

mapping(address => bool) public kycVerified;

event KYCRequirementUpdated(bool required);
event KYCStatusUpdated(address user, bool verified);
```

#### SPEND-01: spendUSDT() in Token Contract

```solidity
// In the USDT wrapper or wallet contract

/// @notice Spend USDT from user's balance for platform purposes
/// @param user User address
/// @param amount Amount to spend (in USDT, 18 decimals)
/// @param purpose Purpose identifier for tracking
function spendUSDT(address user, uint256 amount, string calldata purpose) external {
    require(msg.sender == address(this) || isAdmin(msg.sender), "Unauthorized");
    require(userBalances[user] >= amount, "Insufficient balance");

    userBalances[user] -= amount;
    totalSpent[user] += amount;

    // Transfer to platform or burn
    usdt.transfer(platformWallet, amount);

    emit USDTSpent(user, amount, purpose);
}

/// @notice Internal: called when user purchases NFTs
function _processPurchase(address buyer, uint256 amount, string memory purpose) internal {
    spendUSDT(buyer, amount, purpose);
}

mapping(address => uint256) public totalSpent;

event USDTSpent(address user, uint256 amount, string purpose);
```

```javascript
// BURN-01: Burn NFT
routerAdd("POST", "/api/v2/burn-nft", (e) => {
  try {
    const requestInfo = e.requestInfo()
    const userId = requestInfo.auth?.id

    if (!userId) {
      return e.json(401, {
        success: false,
        error: { message: "Auth required", code: "AUTH_REQUIRED" },
      })
    }

    const body = e.parseBody()
    const nftId = body.nft_id
    const nftType = body.nft_type // "egg", "food", "animal"

    if (!nftId || !nftType) {
      return e.json(400, {
        success: false,
        error: { message: "nft_id and nft_type required", code: "VALIDATION_ERROR" },
      })
    }

    // Get user's wallet
    const user = $app.findRecordById("users", userId)
    const userWallet = user.get("wallet")

    // Verify ownership via wallet-api
    const walletApiUrl = $os.getenv("WALLET_SRV_URL") || "http://localhost:3001"
    const response = $http.send({
      url: `${walletApiUrl}/api/v1/wallet/burn-nft`,
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        user_address: userWallet,
        nft_id: nftId,
        nft_type: nftType,
      }),
    })

    // Parse response
    let responseData
    if (response.json && typeof response.json === "object") {
      responseData = response.json
    } else if (response.body) {
      let responseBody = response.body
      if (Array.isArray(response.body)) {
        responseBody = ""
        for (let i = 0; i < response.body.length; i++) {
          responseBody += String.fromCharCode(response.body[i])
        }
      }
      responseData = JSON.parse(responseBody)
    }

    if (response.statusCode < 200 || response.statusCode >= 300) {
      throw new Error("Burn failed: " + JSON.stringify(responseData))
    }

    // Update PocketBase record (mark as burned)
    let collectionName = ""
    if (nftType === "egg") collectionName = "egg_nfts"
    else if (nftType === "food") collectionName = "food_nfts"
    else if (nftType === "animal") collectionName = "animal_nfts"

    if (collectionName) {
      const record = $app.findRecordById(collectionName, nftId)
      if (record) {
        record.set("is_burned", true)
        record.set("burned_at", new Date().toISOString())
        $app.save(record)
      }
    }

    e.json(200, {
      success: true,
      data: {
        nft_id: nftId,
        nft_type: nftType,
        transaction_hash: responseData.data.transaction_hash,
      },
    })
  } catch (err) {
    e.json(500, { success: false, error: { message: err.message, code: "BURN_NFT_FAILED" } })
  }
})

// SPEND-01: Spend USDT (internal utility)
routerAdd("POST", "/api/v2/spend-usdt", (e) => {
  try {
    const requestInfo = e.requestInfo()
    const userId = requestInfo.auth?.id

    if (!userId) {
      return e.json(401, {
        success: false,
        error: { message: "Auth required", code: "AUTH_REQUIRED" },
      })
    }

    const body = e.parseBody()
    const amount = parseFloat(body.amount)
    const purpose = body.purpose

    if (!amount || !purpose) {
      return e.json(400, {
        success: false,
        error: { message: "amount and purpose required", code: "VALIDATION_ERROR" },
      })
    }

    const user = $app.findRecordById("users", userId)
    const wallet = $app.findFirstRecordByData("user_wallets", "user_id", userId)

    if (!wallet) {
      return e.json(404, {
        success: false,
        error: { message: "Wallet not found", code: "WALLET_NOT_FOUND" },
      })
    }

    const currentBalance = wallet.get("usdt_balance") || 0
    if (currentBalance < amount) {
      return e.json(400, {
        success: false,
        error: { message: "Insufficient balance", code: "INSUFFICIENT_BALANCE" },
      })
    }

    // Update wallet balance
    wallet.set("usdt_balance", currentBalance - amount)
    wallet.set("total_spent", (wallet.get("total_spent") || 0) + amount)
    $app.save(wallet)

    // Log transaction
    const txCollection = $app.getCollection("transaction_logs")
    const txRecord = $app.newRecord(txCollection)
    txRecord.set("user_id", userId)
    txRecord.set("type", "spend")
    txRecord.set("amount", amount)
    txRecord.set("purpose", purpose)
    txRecord.set("created", new Date().toISOString())
    $app.save(txRecord)

    e.json(200, {
      success: true,
      data: {
        amount_spent: amount,
        remaining_balance: currentBalance - amount,
        purpose: purpose,
      },
    })
  } catch (err) {
    e.json(500, { success: false, error: { message: err.message, code: "SPEND_USDT_FAILED" } })
  }
})
```

```javascript
// BURN-01
app.post("/api/v1/wallet/burn-nft", async (req, res) => {
  try {
    const { user_address, nft_id, nft_type } = req.body

    if (!user_address || !nft_id || !nft_type) {
      return res.status(400).json({
        success: false,
        error: { message: "Missing required fields", code: "VALIDATION_ERROR" },
      })
    }

    const user = await pocketBase.collection("users").getFirstListItem(`wallet = "${user_address}"`)
    const privateKey = await decryptPrivateKey(user.encrypted_private_key, MASTER_KEY + user.id)

    const provider = new ethers.JsonRpcProvider(process.env.RPC_URL)
    const signer = new ethers.Wallet(privateKey, provider)

    const eggContract = new ethers.Contract(
      process.env.NEXT_PUBLIC_EGG_NFT_ADDRESS,
      EGG_NFT_ABI,
      signer
    )

    const nftTypeMap = { egg: 0, food: 1, animal: 2 }
    const tx = await eggContract.burnNFT(nft_id, nftTypeMap[nft_type])
    await tx.wait(12)

    res.json({
      success: true,
      data: {
        nft_id: nft_id,
        nft_type: nft_type,
        transaction_hash: tx.hash,
      },
    })
  } catch (error) {
    res.status(500).json({
      success: false,
      error: { message: error.message, code: "BURN_NFT_FAILED" },
    })
  }
})
```

```typescript
import { useState } from "react"
import { pb } from "@/lib/pocketbase/client"
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog"
import { Button } from "@/components/ui/button"
import { AlertTriangle } from "lucide-react"

interface BurnNFTDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  nftId: string
  nftType: "egg" | "food" | "animal"
  onSuccess: () => void
}

export function BurnNFTDialog({ open, onOpenChange, nftId, nftType, onSuccess }: BurnNFTDialogProps) {
  const [burning, setBurning] = useState(false)

  const handleBurn = async () => {
    setBurning(true)
    try {
      await pb.send("/api/v2/burn-nft", {
        body: { nft_id: nftId, nft_type: nftType }
      })
      onSuccess()
      onOpenChange(false)
    } catch (error) {
      console.error("Burn failed:", error)
    } finally {
      setBurning(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2 text-destructive">
            <AlertTriangle className="h-5 w-5" />
            Burn {nftType} NFT
          </DialogTitle>
          <DialogDescription>
            This action is irreversible. The NFT will be permanently destroyed.
            {nftType === "egg" && " Hatched eggs cannot be burned."}
          </DialogDescription>
        </DialogHeader>
        <DialogFooter>
          <Button variant="outline" onClick={() => onOpenChange(false)}>
            Cancel
          </Button>
          <Button variant="destructive" onClick={handleBurn} disabled={burning}>
            {burning ? "Burning..." : "Confirm Burn"}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

#### KYC Status Badge

Create at `apps/web/components/dashboard/KYCStatusBadge.tsx`:

```typescript
import { useState, useEffect } from "react"
import { pb } from "@/lib/pocketbase/client"
import { Shield, ShieldCheck, ShieldX } from "lucide-react"

export function KYCStatusBadge() {
  const [kycStatus, setKycStatus] = useState(null)

  useEffect(() => {
    const fetchStatus = async () => {
      try {
        const response = await pb.send("/api/v2/kyc-status", {})
        setKycStatus(response.data)
      } catch (err) {
        console.error("Failed to fetch KYC status:", err)
      }
    }
    fetchStatus()
  }, [])

  if (!kycStatus) return null

  if (kycStatus.kyc_verified) {
    return (
      <div className="flex items-center gap-2 text-green-600">
        <ShieldCheck className="h-4 w-4" />
        <span className="text-sm">KYC Verified</span>
      </div>
    )
  }

  if (kycStatus.kyc_required_globally) {
    return (
      <div className="flex items-center gap-2 text-destructive">
        <ShieldX className="h-4 w-4" />
        <span className="text-sm">KYC Required — Withdrawals Disabled</span>
      </div>
    )
  }

  return (
    <div className="flex items-center gap-2 text-muted-foreground">
      <Shield className="h-4 w-4" />
      <span className="text-sm">KYC Not Required</span>
    </div>
  )
}
```
