       function requireAdmin(e) {
         const requestInfo = e.requestInfo()
         if (!requestInfo.auth?.id) {
           return e.json(401, { success: false, error: { message: "Authentication required", code: "AUTH_REQUIRED" } })
         }
         
         const user = $app.findRecordById("users", requestInfo.auth.id)
         if (!user) {
           return e.json(401, { success: false, error: { message: "User not found", code: "USER_NOT_FOUND" } })
         }
         
         const role = user.get("role")
         if (role !== "admin") {
           return e.json(403, { success: false, error: { message: "Admin access required", code: "ADMIN_REQUIRED" } })
         }
         
         return user
       }
       ```

         try {
           const adminUser = requireAdmin(e)
           if (!adminUser) return
           
           const body = e.parseBody()
           const action = body.action // "pause" or "unpause"
           
           // Validate action
           if (!["pause", "unpause"].includes(action)) {
             return e.json(400, { success: false, error: { message: "Invalid action", code: "INVALID_ACTION" } })
           }
           
           // Call wallet-api
           const walletApiUrl = $os.getenv("WALLET_SRV_URL") || "http://wallet-api:3001"
           const response = $http.send({
             url: `${walletApiUrl}/api/v1/admin/marketplace-control`,
             method: "POST",
             headers: { "Content-Type": "application/json" },
             body: JSON.stringify({ action }),
           })
           
           if (response.statusCode >= 400) {
             throw new Error("Wallet API call failed")
           }
           
           // Log audit trail
           const auditData = {
             admin_id: adminUser.id,
             action: action,
             timestamp: new Date().toISOString(),
           }
           console.log("[AUDIT] Marketplace control:", auditData)
           
           e.json(200, {
             success: true,
             data: response.json,
           })
         } catch (err) {
           console.error("[MARKETPLACE_CONTROL ERROR]:", err)
           e.json(500, { success: false, error: { message: err.message, code: "CONTROL_FAILED" } })
         }
       })

       app.post("/api/v1/admin/marketplace-control", async (req, res) => {
         try {
           const { action } = req.body
           
           // Validate action
           if (!["pause", "unpause"].includes(action)) {
             return res.status(400).json({ success: false, error: { message: "Invalid action" } })
           }
           
           // Get owner signer
           const ownerPrivateKey = process.env.WALLET_OWNER_PRIVATE_KEY
           if (!ownerPrivateKey) {
             return res.status(500).json({ success: false, error: { message: "Owner key not configured" } })
           }
           
           const provider = new ethers.JsonRpcProvider(process.env.RPC_URL)
           const ownerSigner = new ethers.Wallet(ownerPrivateKey, provider)
           
           // Connect to contract
           const contractAddress = process.env.COMMISSION_DISTRIBUTION_ADDRESS
           const contract = new ethers.Contract(contractAddress, COMMISSION_DISTRIBUTION_ABI, ownerSigner)
           
           // Execute action
           let tx
           if (action === "pause") {
             tx = await contract.pauseMarketplace()
           } else if (action === "unpause") {
             tx = await contract.unpauseMarketplace()
           }
           
           await tx.wait()
           
           // Return transaction hash
           res.json({
             success: true,
             data: {
               transaction_hash: tx.hash,
               action: action,
             },
           })
         } catch (error) {
           console.error("[ADMIN CONTROL ERROR]:", error)
           res.status(500).json({ success: false, error: { message: error.message } })
         }
       })
       ```

<threat_model>

## Trust Boundaries

| Boundary                | Description                                                                    |
| ----------------------- | ------------------------------------------------------------------------------ |
| PocketBase → Wallet-API | Admin user authenticated in PocketBase, calls wallet-api which holds owner key |
| Wallet-API → Blockchain | Owner private key signs transactions, must never leave wallet-api              |

## STRIDE Threat Register

| Threat ID | Category               | Component                         | Disposition | Mitigation Plan                                              |
| --------- | ---------------------- | --------------------------------- | ----------- | ------------------------------------------------------------ |
| T-29-05   | Spoofing               | /api/v2/admin/marketplace-control | mitigate    | requireAdmin checks user role === "admin"                    |
| T-29-06   | Spoofing               | /api/v1/admin/marketplace-control | mitigate    | Wallet-API trusts PocketBase auth (internal network only)    |
| T-29-07   | Information Disclosure | WALLET_OWNER_PRIVATE_KEY          | mitigate    | Key stored in wallet-api .env only, never logged or returned |
| T-29-08   | Tampering              | Audit trail                       | mitigate    | Log admin actions with user ID, timestamp, action type       |

</threat_model>
