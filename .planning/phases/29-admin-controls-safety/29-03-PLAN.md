       ```typescript
       "use client"
       
       import { useState } from "react"
       import { AlertDialog } from "@/components/ui/AlertDialog"
       import { Button } from "@/components/ui/Button"
       import { AlertTriangle, CheckCircle2 } from "lucide-react"
       
       interface PauseControlProps {
         isPaused: boolean
         onPauseChange: () => Promise<void>
       }
       
       export function PauseControl({ isPaused, onPauseChange }: PauseControlProps) {
         const [showConfirm, setShowConfirm] = useState(false)
         const [pending, setPending] = useState(false)
         
         const handlePause = async () => {
           setPending(true)
           try {
             const response = await fetch(`${process.env.NEXT_PUBLIC_POCKETBASE_URL}/api/v2/admin/marketplace-control`, {
               method: "POST",
               headers: {
                 "Content-Type": "application/json",
                 "Authorization": `Bearer ${localStorage.getItem("pb_auth")}`,
               },
               body: JSON.stringify({ action: "pause" }),
             })
             
             if (!response.ok) throw new Error("Failed to pause marketplace")
             await onPauseChange()
           } finally {
             setPending(false)
             setShowConfirm(false)
           }
         }
         
         const handleUnpause = async () => {
           setPending(true)
           try {
             const response = await fetch(`${process.env.NEXT_PUBLIC_POCKETBASE_URL}/api/v2/admin/marketplace-control`, {
               method: "POST",
               headers: {
                 "Content-Type": "application/json",
                 "Authorization": `Bearer ${localStorage.getItem("pb_auth")}`,
               },
               body: JSON.stringify({ action: "unpause" }),
             })
             
             if (!response.ok) throw new Error("Failed to unpause marketplace")
             await onPauseChange()
           } finally {
             setPending(false)
           }
         }
         
         return (
           <div className="clay-card">
             <h3 className="text-lg font-semibold mb-4">Marketplace Controls</h3>
             
             <div className="flex items-center gap-4 mb-6">
               <div className={`flex items-center gap-2 ${isPaused ? "text-red-600" : "text-green-600"}`}>
                 {isPaused ? <AlertTriangle className="w-5 h-5" /> : <CheckCircle2 className="w-5 h-5" />}
                 <span className="font-medium">{isPaused ? "Marketplace Paused" : "Marketplace Active"}</span>
               </div>
             </div>
             
             {isPaused ? (
               <Button onClick={handleUnpause} disabled={pending} variant="primary">
                 {pending ? "Unpausing..." : "Unpause Marketplace"}
               </Button>
             ) : (
               <Button onClick={() => setShowConfirm(true)} disabled={pending} variant="destructive">
                 {pending ? "Pausing..." : "Pause Marketplace"}
               </Button>
             )}
             
             <AlertDialog
               open={showConfirm}
               onOpenChange={setShowConfirm}
               title="Pause Marketplace?"
               description="This will disable all marketplace operations including minting, transfers, and sales. Users will see a pause banner. This action can be reversed at any time."
               onConfirm={handlePause}
               confirmText="Pause Marketplace"
             />
           </div>
         )
       }
       ```
    2. Component receives isPaused state and refresh callback
    3. Uses confirmation modal for pause action (D-10)
    4. Unpause is single-click with no confirmation (D-11)

    1. Create RevenueStats component:
       ```typescript
       "use client"
       
       import { useEffect, useState } from "react"
       import { getPocketBaseClient } from "@/lib/pocketbase/client"
       
       interface RevenueStatsProps {
         className?: string
       }
       
       export function RevenueStats({ className }: RevenueStatsProps) {
         const [totalRevenue, setTotalRevenue] = useState<number>(0)
         const [loading, setLoading] = useState(true)
         const [error, setError] = useState<string | null>(null)
         
         useEffect(() => {
           async function fetchRevenue() {
             try {
               const pb = getPocketBaseClient()
               const records = await pb.collection("transaction_logs").getFullList({
                 filter: 'tx_type = "sale" && status = "completed"',
               })
               
               // Sum USDT amounts
               const total = records.reduce((sum, record) => {
                 return sum + (record.get("amount_usdt") || 0)
               }, 0)
               
               setTotalRevenue(total)
             } catch (err) {
               setError(err instanceof Error ? err.message : "Failed to fetch revenue")
             } finally {
               setLoading(false)
             }
           }
           
           fetchRevenue()
         }, [])
         
         if (loading) {
           return <div className="clay-card">Loading revenue stats...</div>
         }
         
         if (error) {
           return <div className="clay-card text-red-600">Error: {error}</div>
         }
         
         return (
           <div className={`clay-card ${className || ""}`}>
             <h3 className="text-lg font-semibold mb-4">Platform Revenue (USDT)</h3>
             <div className="text-4xl font-bold text-green-600">
               {totalRevenue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })} USDT
             </div>
             <p className="text-sm text-gray-500 mt-2">
               From completed marketplace sales
             </p>
           </div>
         )
       }
       ```
    2. Queries transaction_logs collection (D-09)
    3. Filters by tx_type="sale" and status="completed"
    4. Sums usdt_amount field (D-08)
    5. Displays with claymorphism styling

       ```typescript
       "use client"
       
       import { useEffect, useState } from "react"
       import { AlertTriangle, X } from "lucide-react"
       
       export function PauseBanner() {
         const [isPaused, setIsPaused] = useState(false)
         const [dismissed, setDismissed] = useState(false)
         
         useEffect(() => {
           async function checkPauseStatus() {
             try {
               const response = await fetch(`${process.env.NEXT_PUBLIC_POCKETBASE_URL}/api/v2/admin/marketplace-status`)
               const data = await response.json()
               setIsPaused(data.paused)
             } catch (error) {
               console.error("Failed to check pause status", error)
             }
           }
           
           checkPauseStatus()
         }, [])
         
         if (!isPaused || dismissed) {
           return null
         }
         
         return (
           <div className="bg-red-600 text-white px-4 py-3 flex items-center justify-between gap-4">
             <div className="container mx-auto flex items-center gap-3">
               <AlertTriangle className="w-5 h-5 flex-shrink-0" />
               <p className="font-medium">
                 Marketplace Paused - Trading Disabled
               </p>
             </div>
             <button
               onClick={() => setDismissed(true)}
               className="hover:bg-red-700 p-1 rounded transition-colors"
               aria-label="Dismiss banner"
             >
               <X className="w-5 h-5" />
             </button>
           </div>
         )
       }
       ```
