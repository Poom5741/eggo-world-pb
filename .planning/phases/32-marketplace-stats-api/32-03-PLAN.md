# Plan 32-03: getUserReferralStats() API Endpoint

## Gap: STATS-02

**Problem:** No API endpoint returns user's referral tree statistics (downline count, referral chain, earnings from referrals).

**Goal:** Create PocketBase hook endpoint `/api/v2/referral-stats` that returns the authenticated user's referral statistics.

## Implementation

### 1. PocketBase Hook: Extend `32-market-stats.pb.js`

Add route to existing hook file:

```javascript
routerAdd("GET", "/api/v2/referral-stats", (e) => {
  try {
    const requestInfo = e.requestInfo()
    const userId = requestInfo.auth?.id

    if (!userId) {
      return e.json(401, {
        success: false,
        error: { message: "Authentication required", code: "AUTH_REQUIRED" },
      })
    }

    const user = $app.findRecordById("users", userId)

    // Get user's referral chain (who referred them)
    const referralChain = {
      g1: user.get("upline_G1") || null,
      g2: user.get("upline_G2") || null,
      g3: user.get("upline_G3") || null,
      g4: user.get("upline_G4") || null,
    }

    // Find all users where this user is their G1 (direct recruits)
    const allUsers = $app.findRecordsByFilter("users", "", "-created", 10000, 0)
    let directRecruits = []
    let totalDownline = 0
    let referralEarnings = 0

    for (let i = 0; i < allUsers.length; i++) {
      const u = allUsers[i]
      if (u.id === userId) continue

      const uG1 = u.get("upline_G1")
      const uG2 = u.get("upline_G2")
      const uG3 = u.get("upline_G3")
      const uG4 = u.get("upline_G4")

      // Count direct recruits (G1 = this user)
      if (uG1 === userId) {
        directRecruits.push({
          id: u.id,
          username: u.get("username") || "",
          email: u.get("email") || "",
          created: u.get("created"),
        })
      }

      // Count total downline (any level)
      if (uG1 === userId || uG2 === userId || uG3 === userId || uG4 === userId) {
        totalDownline++
      }
    }

    // Calculate referral earnings from transaction_logs
    const transactions = $app.findRecordsByFilter(
      "transaction_logs",
      `type = 'commission' && (recipient = '${userId}' OR recipient_g1 = '${userId}')`,
      "-created",
      10000,
      0
    )

    for (let i = 0; i < transactions.length; i++) {
      const tx = transactions[i]
      referralEarnings += tx.get("amount") || 0
    }

    e.json(200, {
      success: true,
      data: {
        referral_chain: referralChain,
        direct_recruits: directRecruits,
        direct_recruit_count: directRecruits.length,
        total_downline: totalDownline,
        referral_earnings: referralEarnings,
        timestamp: new Date().toISOString(),
      },
    })
  } catch (err) {
    e.json(500, {
      success: false,
      error: { message: err.message, code: "REFERRAL_STATS_FAILED" },
    })
  }
})
```

### 2. Frontend Component: ReferralStatsPanel

Add to dashboard or create at `apps/web/components/dashboard/ReferralStatsPanel.tsx`:

```typescript
import { useState, useEffect } from "react"
import { pb } from "@/lib/pocketbase/client"
import { Users, DollarSign, Network } from "lucide-react"

interface ReferralStats {
  referral_chain: { g1: string | null; g2: string | null; g3: string | null; g4: string | null }
  direct_recruits: Array<{ id: string; username: string; email: string; created: string }>
  direct_recruit_count: number
  total_downline: number
  referral_earnings: number
}

export function ReferralStatsPanel() {
  const [stats, setStats] = useState<ReferralStats | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const fetchStats = async () => {
      try {
        const response = await pb.send("/api/v2/referral-stats", {})
        setStats(response.data)
      } catch (err) {
        console.error("Failed to fetch referral stats:", err)
      } finally {
        setLoading(false)
      }
    }

    fetchStats()
  }, [])

  if (loading) return <div className="animate-pulse h-48 bg-muted rounded" />
  if (!stats) return null

  return (
    <div className="space-y-6">
      <h2 className="text-xl font-semibold">Referral Statistics</h2>

      <div className="grid grid-cols-3 gap-4">
        <div className="p-4 rounded-lg bg-card border">
          <Users className="h-5 w-5 mb-2 text-primary" />
          <p className="text-sm text-muted-foreground">Direct Recruits</p>
          <p className="text-2xl font-bold">{stats.direct_recruit_count}</p>
        </div>
        <div className="p-4 rounded-lg bg-card border">
          <Network className="h-5 w-5 mb-2 text-primary" />
          <p className="text-sm text-muted-foreground">Total Downline</p>
          <p className="text-2xl font-bold">{stats.total_downline}</p>
        </div>
        <div className="p-4 rounded-lg bg-card border">
          <DollarSign className="h-5 w-5 mb-2 text-primary" />
          <p className="text-sm text-muted-foreground">Referral Earnings</p>
          <p className="text-2xl font-bold">{stats.referral_earnings} USDT</p>
        </div>
      </div>

      {stats.direct_recruits.length > 0 && (
        <div>
          <h3 className="font-medium mb-2">Your Direct Recruits</h3>
          <div className="space-y-2">
            {stats.direct_recruits.map((recruit) => (
              <div key={recruit.id} className="p-3 rounded bg-muted/50 flex justify-between">
                <span>{recruit.username || recruit.email}</span>
                <span className="text-sm text-muted-foreground">
                  {new Date(recruit.created).toLocaleDateString()}
                </span>
              </div>
            ))}
          </div>
        </div>
      )}

      <div>
        <h3 className="font-medium mb-2">Your Referral Chain</h3>
        <div className="flex items-center gap-2 text-sm">
          {stats.referral_chain.g4 && <span className="px-2 py-1 bg-muted rounded">G4</span>}
          {stats.referral_chain.g3 && <span className="px-2 py-1 bg-muted rounded">G3</span>}
          {stats.referral_chain.g2 && <span className="px-2 py-1 bg-muted rounded">G2</span>}
          {stats.referral_chain.g1 && <span className="px-2 py-1 bg-muted rounded">G1</span>}
          <span className="px-2 py-1 bg-primary/10 rounded font-medium">You</span>
        </div>
      </div>
    </div>
  )
}
```
