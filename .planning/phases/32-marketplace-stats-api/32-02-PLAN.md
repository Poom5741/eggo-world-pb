# Plan 32-02: getPlatformStats() API Endpoint

## Gap: STATS-01

**Problem:** No admin API endpoint returns platform-wide statistics (total revenue, total volume, user count).

**Goal:** Create PocketBase hook endpoint `/api/v2/platform-stats` that returns aggregate platform statistics for admin dashboard.

## Implementation

### 1. PocketBase Hook: Extend `32-market-stats.pb.js`

Add route to existing hook file:

```javascript
routerAdd("GET", "/api/v2/platform-stats", (e) => {
  try {
    const requestInfo = e.requestInfo()
    const userId = requestInfo.auth?.id

    if (!userId) {
      return e.json(401, {
        success: false,
        error: { message: "Authentication required", code: "AUTH_REQUIRED" },
      })
    }

    // Verify user is admin
    const user = $app.findRecordById("users", userId)
    const isAdmin = user.get("role") === "admin" || user.get("is_admin") === true

    if (!isAdmin) {
      return e.json(403, {
        success: false,
        error: { message: "Admin access required", code: "ADMIN_REQUIRED" },
      })
    }

    // Count total users
    const users = $app.getCollection("users")
    const totalUsers = $app.findRecordsByFilter("users", "", "-created", 1, 0).length
    // Note: PocketBase doesn't have COUNT, so we fetch page size and estimate
    // For accurate count, use pagination or add a stats collection

    // Count total NFTs minted
    const eggNfts = $app.findRecordsByFilter("egg_nfts", "", "-created", 1, 0).length
    const animalNfts = $app.findRecordsByFilter("animal_nfts", "", "-created", 1, 0).length
    const foodNfts = $app.findRecordsByFilter("food_nfts", "", "-created", 1, 0).length

    // Get all transaction logs for totals
    const allTransactions = $app.findRecordsByFilter("transaction_logs", "", "-created", 10000, 0)

    let totalRevenue = 0
    let totalVolume = 0
    let totalCommissionsPaid = 0
    let coinstorBalance = 0

    for (let i = 0; i < allTransactions.length; i++) {
      const tx = allTransactions[i]
      const amount = tx.get("amount") || 0
      const type = tx.get("type")

      if (type === "sale" || type === "mint") {
        totalRevenue += amount
      }
      if (type === "sale") {
        totalVolume += amount
      }
      if (type === "commission") {
        totalCommissionsPaid += amount
      }
      if (type === "coinstor_deposit") {
        coinstorBalance += amount
      }
    }

    e.json(200, {
      success: true,
      data: {
        total_users: totalUsers,
        total_eggs_minted: eggNfts,
        total_animals_hatched: animalNfts,
        total_food_minted: foodNfts,
        total_revenue: totalRevenue,
        total_volume: totalVolume,
        total_commissions_paid: totalCommissionsPaid,
        coinstor_balance: coinstorBalance,
        timestamp: new Date().toISOString(),
      },
    })
  } catch (err) {
    e.json(500, {
      success: false,
      error: { message: err.message, code: "PLATFORM_STATS_FAILED" },
    })
  }
})
```

### 2. Admin Page: `/admin/stats`

Create or update admin stats page at `apps/web/app/admin/stats/page.tsx`:

```typescript
"use client"

import { useState, useEffect } from "react"
import { pb } from "@/lib/pocketbase/client"
import { useIsHydrated } from "@/lib/hooks/use-is-hydrated"

interface PlatformStats {
  total_users: number
  total_eggs_minted: number
  total_animals_hatched: number
  total_food_minted: number
  total_revenue: number
  total_volume: number
  total_commissions_paid: number
  coinstor_balance: number
}

export default function AdminStatsPage() {
  const isHydrated = useIsHydrated()
  const [stats, setStats] = useState<PlatformStats | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!isHydrated) return

    const fetchStats = async () => {
      try {
        const response = await pb.send("/api/v2/platform-stats", {})
        setStats(response.data)
      } catch (err) {
        console.error("Failed to fetch platform stats:", err)
      } finally {
        setLoading(false)
      }
    }

    fetchStats()
  }, [isHydrated])

  if (!isHydrated || loading) {
    return <div className="p-8">Loading...</div>
  }

  if (!stats) {
    return <div className="p-8 text-destructive">Failed to load stats</div>
  }

  return (
    <div className="p-8 max-w-7xl mx-auto">
      <h1 className="text-3xl font-bold mb-8">Platform Statistics</h1>

      <div className="grid grid-cols-2 md:grid-cols-4 gap-6">
        <StatCard label="Total Users" value={stats.total_users} />
        <StatCard label="Eggs Minted" value={stats.total_eggs_minted} />
        <StatCard label="Animals Hatched" value={stats.total_animals_hatched} />
        <StatCard label="Food Minted" value={stats.total_food_minted} />
        <StatCard label="Total Revenue" value={`${stats.total_revenue} USDT`} />
        <StatCard label="Total Volume" value={`${stats.total_volume} USDT`} />
        <StatCard label="Commissions Paid" value={`${stats.total_commissions_paid} USDT`} />
        <StatCard label="CoinStor Balance" value={`${stats.coinstor_balance} USDT`} />
      </div>
    </div>
  )
}

function StatCard({ label, value }: { label: string; value: string | number }) {
  return (
    <div className="p-6 rounded-lg bg-card border">
      <p className="text-sm text-muted-foreground mb-2">{label}</p>
      <p className="text-2xl font-bold">{value}</p>
    </div>
  )
}
```

## Acceptance Criteria

- [ ] GET `/api/v2/platform-stats` requires authentication and admin role

- [ ] Returns aggregate stats: users, NFTs, revenue, volume, commissions, CoinStor
- [ ] Admin page at `/admin/stats` displays all metrics in grid layout
- [ ] Hydration-safe with `useIsHydrated()` hook
