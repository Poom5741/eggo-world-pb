       - Model hook (database operations) → use onModelBeforeCreate, etc.

    2. Convert the pattern:

    **Example conversion for collection hook:**
    ```javascript
    // ❌ BEFORE (CommonJS - fails)
    module.exports = async (e) => {
      const { Dao } = require('pocketbase')
      // logic here
    }

    // ✅ AFTER (PocketBase native - works)
    onRecordCreate("collection_name", (e) => {
      // logic here (Dao is globally available, no require needed)
      e.next()
    })
    ```

    **Example conversion for router endpoint:**
    ```javascript
    // ❌ BEFORE (CommonJS - fails)
    module.exports = async (e) => {
      e.json(200, { success: true })
    }

    // ✅ AFTER (PocketBase native - works)
    routerAdd("POST", "/api/endpoint", (e) => {
      e.json(200, { success: true })
    })
    ```

    migrations.push({
      id: '1774772604_add_referral_chain_field',
      up: async (db) => {
        const collection = await db.findCollectionByNameOrId("users")
        // Add field to collection schema

          type: 'json',
          required: false,
          hidden: true

        await db.saveCollection(collection)

        await db.saveCollection(collection)
      }
