From apps/backend/pb_hooks/01-create-wallet.pb.js (updated in 04-02):

```javascript
// Generate 20-char random password
const charset = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%^&*"
let passwordSecretkey = ""
for (let i = 0; i < 20; i++) {
  passwordSecretkey += charset.charAt(Math.floor(Math.random() * charset.length))
}

// Call wallet API
const response = $http.send({
  url: EGGO_CONFIG.wallet.srvUrl + "/api/wallet/create",
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ passwordSecretkey, publicEncryption: false }),
})

// Parse response (handle byte array to string conversion)
let responseBody = response.body
if (typeof responseBody === "object" && responseBody.length !== undefined) {
  responseBody = new TextDecoder().decode(responseBody)
}
const responseData = JSON.parse(responseBody)

// Save to user record
e.record.set("wallet", responseData.data.address)
e.record.set("pin", passwordSecretkey)
e.record.set("daccPublickey", responseData.data.daccPublickey)
```
