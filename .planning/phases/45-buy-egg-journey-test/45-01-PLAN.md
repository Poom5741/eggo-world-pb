```typescript
test.describe("Buy Egg Journey", () => {
  test.describe.configure({ mode: "serial" })

  // Main journey test per D-01, D-17
  test("full buy journey - authenticated user purchases egg and verifies ownership", async ({
    page,
  }) => {
    // Step 1: Login via E2E bypass (per D-04)
    await e2eLogin(page, "test_buyer", "/marketplace")

    // Step 2: Browse marketplace - verify listings visible
    await expect(page.locator('[data-testid="listing-card"]').first()).toBeVisible()

    // Step 3: Click on first egg listing to view details
    // Note: Using pre-created listing from PocketBase (per D-09, D-12)
    await page.click('[data-testid="listing-card"] >> nth=0')
    await page.waitForURL(/marketplace\/detail/)

    // Step 4: Click Buy Now button
    await page.click('button:has-text("Buy")')

    // Step 5: Confirm purchase in dialog
    await page.click('button:has-text("Confirm Purchase")')

    // Step 6: Wait for success toast or redirect
    await page.waitForURL(/eggs|inventory/, { timeout: 30000 })

    // Step 7: Triple verification (per D-05, D-06)
    // Get tokenId from response or page content
    const tokenId = await extractTokenIdFromPage(page)
    const result = await verifyEggOwnership(
      page,
      tokenId,
      TEST_USERS.test_buyer.walletAddress,
      userId
    )

    expect(result.uiVisible).toBe(true)
    expect(result.onChainOwner.toLowerCase()).toBe(
      TEST_USERS.test_buyer.walletAddress.toLowerCase()
    )
    expect(result.allMatch).toBe(true)
  })

  // Error scenario per D-13, D-14
  test("insufficient balance shows error - no transaction", async ({ page }) => {
    await e2eLogin(page, "test_buyer_poor", "/marketplace")
    await page.click('[data-testid="listing-card"] >> nth=0')
    await page.click('button:has-text("Buy")')
    await page.click('button:has-text("Confirm Purchase")')

    // Per D-15: Error toast visible, no redirect
    await expect(page.locator('[data-sonner-toast][data-type="error"]')).toBeVisible()
    await expect(page).toHaveURL(/marketplace/) // Still on marketplace
  })
})
```
