    3. Component structure:
       ```typescript
       export function ListForSaleModal({ nftId, onSuccess }: ListForSaleModalProps) {
         const [open, setOpen] = useState(false)
         const [price, setPrice] = useState('')
         const [loading, setLoading] = useState(false)
         const [error, setError] = useState<string | null>(null)
       
         const handleList = async () => {
           setLoading(true)
           setError(null)
           
           try {
             const priceNum = parseFloat(price)
             if (isNaN(priceNum) || priceNum <= 0) {
               throw new Error('Invalid price')
             }
             
             // Update NFT in PocketBase
             await pb.collection('nfts').update(nftId, {
               is_listed: true,
               listed_price: priceNum
             })
             
             setOpen(false)
             setPrice('')
             onSuccess?.()
           } catch (err) {
             setError(err.message)
           } finally {
             setLoading(false)
           }
         }
       
         return (
           <Dialog open={open} onOpenChange={setOpen}>
             <DialogTrigger asChild>
               <Button>List for Sale</Button>
             </DialogTrigger>
             <DialogContent>
               <DialogHeader>
                 <DialogTitle>List NFT for Sale</DialogTitle>
               </DialogHeader>
               <div className="space-y-4">
                 <div className="space-y-2">
                   <Label htmlFor="price">Price (USDT)</Label>
                   <Input
                     id="price"
                     type="number"
                     step="0.01"
                     value={price}
                     onChange={(e) => setPrice(e.target.value)}
                     placeholder="0.00"
                   />
                 </div>
                 
                 {error && (
                   <Alert variant="destructive">
                     <AlertDescription>{error}</AlertDescription>
                   </Alert>
                 )}
                 
                 <Button onClick={handleList} disabled={loading || !price}>
                   {loading ? 'Listing...' : 'List for Sale'}
                 </Button>
               </div>
             </DialogContent>
           </Dialog>
         )
       }
       ```
