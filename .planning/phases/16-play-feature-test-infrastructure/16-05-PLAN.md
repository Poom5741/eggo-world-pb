    })
  })

  it("sets canClaim to true when 24 hours passed", async () => {
    const { result } = renderHook(() => useDailyCheckin())

    await waitFor(() => {
      expect(result.current.canClaim).toBe(true)

  it("claimCheckin calls API and updates state", async () => {
