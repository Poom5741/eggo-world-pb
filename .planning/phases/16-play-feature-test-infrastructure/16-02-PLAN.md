```typescript
export function useDailyCheckin(): UseDailyCheckinReturn {
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [streak, setStreak] = useState(0)
  const [lastCheckIn, setLastCheckIn] = useState<string | null>(null)
  const [canClaim, setCanClaim] = useState(false)
  const [countdown, setCountdown] = useState("00:00:00")

  const refreshCheckinStatus = useCallback(async () => {
    try {
      const pb = createClient()
      const user = pb.authStore.record
      if (!user) return

      const userStats = await pb.collection("user_stats").getFirstListItem(`user = "${user.id}"`)

      const lastCheckInTime = userStats.last_check_in || null
      setLastCheckIn(lastCheckInTime)
      setStreak(userStats.check_in_streak || 0)

      // Check if 24 hours passed since last check-in
      if (lastCheckInTime) {
        const lastTime = new Date(lastCheckInTime).getTime()
        const now = Date.now()
        const hoursPassed = (now - lastTime) / (1000 * 60 * 60)
        setCanClaim(hoursPassed >= 24)
      } else {
        // Never checked in before
        setCanClaim(true)
      }
    } catch (err: any) {
      setError(err.message)
    }
  }, [])

  const claimCheckin = useCallback(async (): Promise<boolean> => {
    setLoading(true)
    setError(null)
    try {
      const pb = createClient()
      const response = await pb.send("/api/v2/check-in", {
        method: "POST",
        body: {},
      })

      if (response.success) {
        setStreak(response.data.streak)
        setLastCheckIn(response.data.last_check_in)
        setCanClaim(false)
        await refreshCheckinStatus()
        return true
      } else {
        setError(response.error?.message || "Check-in failed")
        return false
      }
    } catch (err: any) {
      setError(err.message)
      return false
    } finally {
      setLoading(false)
    }
  }, [refreshCheckinStatus])

  // Countdown timer effect (updates every second)
  useEffect(() => {
    if (!lastCheckIn || canClaim) {
      setCountdown("00:00:00")
      return
    }

    const interval = setInterval(() => {
      const lastTime = new Date(lastCheckIn).getTime()
      const nextClaimTime = lastTime + 24 * 60 * 60 * 1000
      const now = Date.now()
      const diff = nextClaimTime - now

      if (diff <= 0) {
        setCountdown("00:00:00")
        setCanClaim(true)
        clearInterval(interval)
        return
      }

      const hours = Math.floor(diff / (1000 * 60 * 60))
      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
      const seconds = Math.floor((diff % (1000 * 60)) / 1000)

      setCountdown(
        `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`
      )
    }, 1000)

    return () => clearInterval(interval)
  }, [lastCheckIn, canClaim])

  // Load check-in status on mount
  useEffect(() => {
    refreshCheckinStatus()
  }, [refreshCheckinStatus])

  return {
    loading,
    error,
    streak,
    lastCheckIn,
    canClaim,
    countdown,
    claimCheckin,
    refreshCheckinStatus,
  }
}
```

**Critical rules:**

- Use `pb.send("/api/v2/check-in", ...)` NOT `pb.collection().create()` (custom endpoint, D-16)
- Countdown updates every 1 second with setInterval
- Countdown format: "HH:MM:SS" with zero-padding (e.g., "08:05:03")
- Check canClaim: 24 hours since last_check_in timestamp
- Streak from user_stats.check_in_streak field (D-15)
- Hook calls refreshCheckinStatus on mount

"use client"

import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogDescription,
  DialogFooter,
} from "@/components/ui/dialog"
import { Button } from "@/components/ui/button"
import { Badge } from "@/components/ui/badge"
import { useDailyCheckin } from "@/hooks/use-daily-checkin"

interface CheckInDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  onSuccess?: () => void
}

function getStreakColor(streak: number): string {
  if (streak >= 14) return "text-warning"
  if (streak >= 7) return "text-primary"
  if (streak >= 4) return "text-on-surface"
  return "text-on-surface-variant"
}

export function CheckInDialog({ open, onOpenChange, onSuccess }: CheckInDialogProps) {
  const { loading, error, streak, canClaim, countdown, claimCheckin } = useDailyCheckin()

  const handleClaim = async () => {
    const success = await claimCheckin()
    if (success) {
      onSuccess?.()
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent variant="clay" className="max-w-md">
        <DialogHeader variant="clay">
          <DialogTitle variant="clay">Daily Check-In</DialogTitle>
          <DialogDescription variant="clay">
            Claim your daily food NFT reward!
          </DialogDescription>
        </DialogHeader>

        {/* Streak display (D-03, D-04) */}
        <div className="flex items-center justify-center py-4">
          <div className="text-center">
            <p className={`text-4xl font-black ${getStreakColor(streak)}`}>
              {streak}d 🔥
            </p>
            <p className="text-sm text-on-surface-variant mt-1">Current Streak</p>
          </div>
        </div>

        {/* Milestone badges */}
        <div className="flex justify-center gap-2 mb-4">
          {streak >= 7 && (
            <Badge variant="clay" className="bg-primary text-on-primary">
              7-Day Bonus ✓
            </Badge>
          )}
          {streak >= 30 && (
            <Badge variant="clay" className="bg-warning text-on-warning">
              30-Day Bonus ✓
            </Badge>
          )}
        </div>

        {/* Countdown or claim button */}
        <div className="text-center py-4">
          {canClaim ? (
            <Button
              onClick={handleClaim}
              disabled={loading}
              variant="clay"
              size="clay-lg"
              className="w-full min-h-[44px]"
            >
              {loading ? (
                <>
                  <span className="material-symbols-outlined animate-spin">sync</span>
                  Claiming...
                </>
              ) : (
                <>
                  <span className="material-symbols-outlined">card_giftcard</span>
                  Claim 1 Food NFT
                </>
              )}
            </Button>
          ) : (
            <div className="space-y-2">
              <p className="text-sm text-on-surface-variant">Next check-in in:</p>
              <p className="text-3xl font-black text-on-surface font-mono">
                {countdown}
              </p>
            </div>
          )}
        </div>

        {/* Error message */}
        {error && (
          <div className="bg-error-container p-3 rounded-lg">
            <p className="text-sm text-on-error-container">{error}</p>
          </div>
        )}

        <DialogFooter variant="clay">
          <Button
            onClick={() => onOpenChange(false)}
            variant="clay-secondary"
            size="clay-lg"
            className="w-full min-h-[44px]"
          >
            Close
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
