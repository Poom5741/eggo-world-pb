```typescript
"use client"

import { useState, useEffect, useCallback } from "react"
import { createClient } from "@/lib/pocketbase/client"

export interface Transaction {
  id: string
  type: "mint" | "transfer" | "claim" | "check_in"
  amount: number
  status: "confirmed" | "pending"
  timestamp: string
  tx_hash?: string
  description: string
}

interface UseTransactionHistoryReturn {
  transactions: Transaction[]
  loading: boolean
  error: string | null
  refresh: () => Promise<void>
}

export function useTransactionHistory(): UseTransactionHistoryReturn {
  const [transactions, setTransactions] = useState<Transaction[]>([])
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const refresh = useCallback(async () => {
    try {
      setLoading(true)
      setError(null)

      const pb = createClient()
      const user = pb.authStore.record
      if (!user) return

      const records = await pb.collection("transactions").getList(1, 10, {
        filter: `user = "${user.id}"`,
        sort: "-created",
      })

      const mappedTransactions: Transaction[] = records.items.map((record: any) => ({
        id: record.id,
        type: record.getString("type") || "transfer",
        amount: record.getNumber("amount") || 0,
        status: record.getString("status") || "pending",
        timestamp: record.getString("created") || record.updated,
        tx_hash: record.getString("tx_hash") || undefined,
        description: record.getString("description") || `${record.getString("type")} transaction`,
      }))

      setTransactions(mappedTransactions)
    } catch (err: any) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    refresh()
  }, [refresh])

  return {
    transactions,
    loading,
    error,
    refresh,
  }
}
```

**Critical rules:**

- Use `pb.collection("transactions").getList()` to fetch last 10 records
- Sort by `-created` (newest first)
- Filter by authenticated user: `user = "${user.id}"`
- Map PocketBase record fields to Transaction interface
- Handle missing fields with defaults (type: "transfer", status: "pending")
- Hook refreshes on mount

```typescript
  DialogTitle,
  DialogDescription,

import { Badge } from "@/components/ui/badge"
import { useWalletBalance } from "@/hooks/use-wallet-balance"
import { useTransactionHistory, Transaction } from "@/hooks/use-transaction-history"

interface BalanceModalProps {
  open: boolean
  onOpenChange: (open: boolean) => void
}

function getTransactionIcon(type: string): string {
  switch (type) {
    case "mint":
      return "add_circle"
    case "claim":
    case "check_in":
      return "download"
    case "transfer":
    default:
      return "swap_horiz"
  }
}

function getTransactionColor(type: string): string {
  switch (type) {
    case "mint":
      return "text-primary"
    case "claim":
    case "check_in":
      return "text-tertiary"
    case "transfer":
    default:
      return "text-secondary"
  }
}

function formatTimestamp(timestamp: string): string {
  const date = new Date(timestamp)
  return date.toLocaleDateString("en-US", {
    month: "short",
    day: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  })
}

export function BalanceModal({ open, onOpenChange }: BalanceModalProps) {
  const { balance, loading: balanceLoading } = useWalletBalance()
  const { transactions, loading: txLoading, refresh } = useTransactionHistory()

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent variant="clay" className="max-w-md max-h-[90vh] flex flex-col">
        <DialogHeader variant="clay">
          <DialogTitle variant="clay">Wallet Balance</DialogTitle>
          <DialogDescription variant="clay">

        </DialogHeader>

        {/* Balance Breakdown (D-12) */}

                <p className="text-sm text-on-surface-variant">USDT Balance</p>
              </div>
              <p className="text-2xl font-black text-on-surface">
                {balanceLoading ? "..." : `$${(balance?.usdt_balance || 0).toFixed(2)}`}
              </p>
            </div>
          </div>

          <div className="bg-surface-container p-4 rounded-lg">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2">
                <span className="material-symbols-outlined text-warning">schedule</span>

        <div className="flex-1 overflow-y-auto">
```
