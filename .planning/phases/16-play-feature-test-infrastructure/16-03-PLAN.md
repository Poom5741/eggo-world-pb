**Complete implementation:**

```javascript
/**
 * Hook: 17-claim-checkin.pb.js
 * Event: Router (POST /api/v2/check-in)
 *
 * Flow:
 * 1. Authenticate user
 * 2. Get or create user_stats record
 * 3. Validate 24-hour cooldown since last_check_in
 * 4. Calculate new streak (reset if missed day)
 * 5. Determine reward based on streak milestone
 * 6. Mint Food NFT(s) via wallet-api
 * 7. Update user_stats with new streak and timestamp
 * 8. Return success with reward details
 *
 * Request Body: {} (empty, user from auth)
 *
 * Response:
 * {
 *   "success": true,
 *   "data": {
 *     "streak": 14,
 *     "reward_count": 1,
 *     "milestone": null,
 *     "last_check_in": "2026-04-19T10:30:00.000Z",
 *     "next_check_in": "2026-04-20T10:30:00.000Z"
 *   }
 * }
 */

var WALLET_SRV_URL = $os.getenv("WALLET_SRV_URL") || "http://wallet-api:3001"

routerAdd("POST", "/api/v2/check-in", (e) => {
  try {
    // Step 1: Authenticate user
    const user = $apis.requireAuth(e)

    // Step 2: Get or create user_stats record
    let userStats
    try {
      userStats = $app.dao().findFirstRecordByData("user_stats", "user", user.id)
    } catch (err) {
      // Create user_stats if doesn't exist
      const statsCollection = $app.dao().findCollectionByNameOrId("user_stats")
      userStats = new Record(statsCollection)
      userStats.set("user", user.id)
      userStats.set("check_in_streak", 0)
      userStats.set("last_check_in", null)
      userStats.set("check_in_count", 0)
      $app.dao().saveRecord(userStats)
    }

    // Step 3: Validate 24-hour cooldown
    const lastCheckIn = userStats.getString("last_check_in")
    if (lastCheckIn) {
      const lastTime = new Date(lastCheckIn).getTime()
      const now = Date.now()
      const hoursSinceLastCheckIn = (now - lastTime) / (1000 * 60 * 60)

      if (hoursSinceLastCheckIn < 24) {
        const hoursRemaining = 24 - hoursSinceLastCheckIn
        return e.json(400, {
          success: false,
          error: {
            message: `Check-in not available yet. Please wait ${Math.ceil(hoursRemaining)} more hours.`,
            code: "COOLDOWN_ACTIVE",
            hours_remaining: hoursRemaining.toFixed(2),
          },
        })
      }

      // Step 4: Calculate new streak
      const daysSinceLastCheckIn = hoursSinceLastCheckIn / 24
      if (daysSinceLastCheckIn > 1.5) {
        // Missed a day (more than 36 hours), reset streak
        userStats.set("check_in_streak", 0)
      }
    }

    // Increment streak
    const newStreak = (userStats.getInt("check_in_streak") || 0) + 1
    userStats.set("check_in_streak", newStreak)

    // Step 5: Determine reward based on streak milestone
    let rewardCount = 1
    let milestone = null

    if (newStreak === 30) {
      rewardCount = 5
      milestone = "30_day"
    } else if (newStreak === 7 || (newStreak > 7 && newStreak % 7 === 0)) {
      rewardCount = 2
      milestone = "7_day"
    }

    // Step 6: Mint Food NFT(s) via wallet-api
    const userAddress = user.getString("wallet")
    if (!userAddress) {
      return e.json(400, {
        success: false,
        error: {
          message: "User wallet address not found",
          code: "WALLET_NOT_FOUND",
        },
      })
    }

    // Mint Food NFTs (one at a time)
    const mintedTokens = []
    for (let i = 0; i < rewardCount; i++) {
      try {
        const mintResponse = $http.send({
          url: WALLET_SRV_URL + "/mint-food",
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            user_address: userAddress,
            food_type: "grain", // Default food type
            quantity: 1,
          }),
        })

        const mintResult = JSON.parse(mintResponse.content)
        if (mintResult.success && mintResult.data && mintResult.data.token_id) {
          mintedTokens.push(mintResult.data.token_id)
        }
      } catch (mintError) {
        console.error("Failed to mint Food NFT #" + (i + 1) + ":", mintError)
        // Continue with remaining mints even if one fails
      }
    }

    // Step 7: Update user_stats
    userStats.set("last_check_in", new Date().toISOString())
    userStats.set("check_in_count", (userStats.getInt("check_in_count") || 0) + 1)
    $app.dao().saveRecord(userStats)

    // IMPORTANT: Call e.next() to commit the record
    e.next()

    // Step 8: Return success
    const nextCheckIn = new Date()
    nextCheckIn.setHours(nextCheckIn.getHours() + 24)

    e.json(200, {
      success: true,
      data: {
        streak: newStreak,
        reward_count: rewardCount,
        milestone: milestone,
        minted_tokens: mintedTokens,
        last_check_in: userStats.getString("last_check_in"),
        next_check_in: nextCheckIn.toISOString(),
      },
    })
  } catch (error) {
    console.error("Check-in error:", error)
    e.json(500, {
      success: false,
      error: {
        message: error.message || "Check-in failed",
        code: "CHECKIN_FAILED",
      },
    })
  }
})

console.log("Check-in endpoint registered at POST /api/v2/check-in")
```

**Critical rules:**

- Use `$apis.requireAuth(e)` for authentication (NOT `e.requireAuth()`)
- Call `e.next()` AFTER `$app.dao().saveRecord(userStats)` to commit changes (Key Learning #2 from STATE.md)
- Use `$http.send()` for wallet-api calls (NOT fetch)
- Validate 24-hour cooldown: reject if < 24 hours since last_check_in (D-19)
- Reset streak if > 36 hours since last check-in (missed a day, D-20)
- Reward structure: 1 NFT daily, 2 NFTs at 7-day, 5 NFTs at 30-day (D-18)
- Handle wallet-api failures gracefully (log error, continue with remaining mints)
- Default food_type is "grain" for check-in rewards
- Use WALLET_SRV_URL environment variable (NOT hardcoded URL, Key Learning #4)
- Mint NFTs one at a time in loop (wallet-api expects quantity: 1 per call)
- Return detailed response with streak, reward_count, milestone, minted_tokens
