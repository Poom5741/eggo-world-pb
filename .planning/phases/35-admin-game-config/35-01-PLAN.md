# Plan 35-01: Admin Game Configuration Functions

## Gaps: ADMIN-01, ADMIN-02, ADMIN-03, ADMIN-04

**Problem:** Game parameters (platform fee, breed cooldown, rarity weights, species catalog) are hardcoded in smart contracts. Admins cannot adjust them without redeploying contracts.

**Goal:** Implement 4 admin-only configuration functions that allow dynamic game parameter adjustment.

## Implementation

### 1. Smart Contract: AdminConfig.sol

Create new contract or extend existing EggNFT.sol:

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/security/Pausable.sol";

contract AdminConfig is Ownable, Pausable {
    // ADMIN-01: Platform fee percentage (basis points: 100 = 1%)
    uint256 public platformFeePercent = 400; // 4% default

    // ADMIN-02: Breed cooldown in seconds
    uint256 public breedCooldown = 172800; // 48 hours default

    // ADMIN-03: Rarity weights (basis points)
    struct RarityWeights {
        uint256 common;    // 6000 = 60%
        uint256 rare;      // 2500 = 25%
        uint256 epic;      // 1200 = 12%
        uint256 legendary; // 300 = 3%
    }

    RarityWeights public rarityWeights;

    // ADMIN-04: Species catalog
    struct Species {
        uint256 id;
        string name;
        bool active;
        uint256 weight; // Probability weight
    }

    Species[] public speciesList;
    mapping(uint256 => bool) public speciesExists;

    // Events
    event PlatformFeeUpdated(uint256 oldFee, uint256 newFee);
    event BreedCooldownUpdated(uint256 oldCooldown, uint256 newCooldown);
    event RarityWeightsUpdated(uint256 common, uint256 rare, uint256 epic, uint256 legendary);
    event SpeciesAdded(uint256 speciesId, string name);
    event SpeciesToggled(uint256 speciesId, bool active);

    constructor() {
        // Initialize default rarity weights
        rarityWeights = RarityWeights(6000, 2500, 1200, 300);

        // Initialize default species
        _addSpecies(1, "Dragon", true, 100);
        _addSpecies(2, "Wolf", true, 150);
        _addSpecies(3, "Phoenix", true, 80);
        _addSpecies(4, "Cat", true, 200);
        _addSpecies(5, "Fox", true, 170);
    }

    // ========== ADMIN-01: setPlatformFee ==========

    /// @notice Set platform fee percentage (in basis points)
    /// @param feePercent Fee in basis points (100 = 1%, max 2000 = 20%)
    function setPlatformFee(uint256 feePercent) external onlyOwner {
        require(feePercent <= 2000, "Fee cannot exceed 20%");

        uint256 oldFee = platformFeePercent;
        platformFeePercent = feePercent;

        emit PlatformFeeUpdated(oldFee, feePercent);
    }

    // ========== ADMIN-02: setBreedCooldown ==========

    /// @notice Set breeding cooldown duration
    /// @param cooldownSeconds Cooldown in seconds (min 3600 = 1 hour, max 604800 = 7 days)
    function setBreedCooldown(uint256 cooldownSeconds) external onlyOwner {
        require(cooldownCooldown >= 3600, "Cooldown must be at least 1 hour");
        require(cooldownSeconds <= 604800, "Cooldown cannot exceed 7 days");

        uint256 oldCooldown = breedCooldown;
        breedCooldown = cooldownSeconds;

        emit BreedCooldownUpdated(oldCooldown, cooldownSeconds);
    }

    // ========== ADMIN-03: updateRarityWeights ==========

    /// @notice Update rarity distribution weights
    /// @param weights New rarity weights (must sum to 10000)
    function updateRarityWeights(RarityWeights calldata weights) external onlyOwner {
        uint256 total = weights.common + weights.rare + weights.epic + weights.legendary;
        require(total == 10000, "Weights must sum to 10000 (100%)");

        rarityWeights = weights;

        emit RarityWeightsUpdated(weights.common, weights.rare, weights.epic, weights.legendary);
    }

    /// @notice Get current rarity distribution as percentages
    function getRarityWeights() external view returns (
        uint256 commonPct,
        uint256 rarePct,
        uint256 epicPct,
        uint256 legendaryPct
    ) {
        return (
            rarityWeights.common / 100,
            rarityWeights.rare / 100,
            rarityWeights.epic / 100,
            rarityWeights.legendary / 100
        );
    }

    // ========== ADMIN-04: addNewSpecies ==========

    /// @notice Add new species to catalog
    /// @param speciesId Unique species ID
    /// @param name Species display name
    /// @param weight Probability weight (higher = more common)
    function addNewSpecies(uint256 speciesId, string calldata name, uint256 weight) external onlyOwner {
        require(!speciesExists[speciesId], "Species already exists");
        require(weight > 0, "Weight must be positive");
        require(bytes(name).length > 0, "Name cannot be empty");

        _addSpecies(speciesId, name, true, weight);
    }

    /// @notice Toggle species active status
    /// @param speciesId Species to toggle
    /// @param active Whether species should be active
    function toggleSpecies(uint256 speciesId, bool active) external onlyOwner {
        require(speciesExists[speciesId], "Species does not exist");

        for (uint256 i = 0; i < speciesList.length; i++) {
            if (speciesList[i].id == speciesId) {
                speciesList[i].active = active;
                emit SpeciesToggled(speciesId, active);
                break;
            }
        }
    }

    /// @notice Get all active species
    function getActiveSpecies() external view returns (Species[] memory) {
        uint256 activeCount = 0;
        for (uint256 i = 0; i < speciesList.length; i++) {
            if (speciesList[i].active) {
                activeCount++;
            }
        }

        Species[] memory result = new Species[](activeCount);
        uint256 index = 0;
        for (uint256 i = 0; i < speciesList.length; i++) {
            if (speciesList[i].active) {
                result[index] = speciesList[i];
                index++;
            }
        }

        return result;
    }

    // Internal helpers
    function _addSpecies(uint256 speciesId, string memory name, bool active, uint256 weight) internal {
        speciesList.push(Species(speciesId, name, active, weight));
        speciesExists[speciesId] = true;
        emit SpeciesAdded(speciesId, name);
    }
}
```

```javascript
// ADMIN-01: Set platform fee
routerAdd("POST", "/api/v2/admin/set-platform-fee", (e) => {
  try {
    const requestInfo = e.requestInfo()
    const userId = requestInfo.auth?.id

    if (!userId) {
      return e.json(401, {
        success: false,
        error: { message: "Auth required", code: "AUTH_REQUIRED" },
      })
    }

    const user = $app.findRecordById("users", userId)
    if (user.get("role") !== "admin") {
      return e.json(403, {
        success: false,
        error: { message: "Admin required", code: "ADMIN_REQUIRED" },
      })
    }

    const body = e.parseBody()
    const feePercent = parseInt(body.fee_percent)

    if (isNaN(feePercent) || feePercent < 0 || feePercent > 2000) {
      return e.json(400, {
        success: false,
        error: { message: "Fee must be 0-2000 basis points", code: "VALIDATION_ERROR" },
      })
    }

    // Call wallet-api
    const walletApiUrl = $os.getenv("WALLET_SRV_URL") || "http://localhost:3001"
    const response = $http.send({
      url: `${walletApiUrl}/api/v1/wallet/admin/set-platform-fee`,
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ fee_percent: feePercent }),
    })

    e.json(200, { success: true, data: { fee_percent: feePercent } })
  } catch (err) {
    e.json(500, { success: false, error: { message: err.message, code: "SET_FEE_FAILED" } })
  }
})
```

## Acceptance Criteria

- [ ] ADMIN-01: `setPlatformFee()` updates fee with validation (0-2000 basis points)
- [ ] ADMIN-02: `setBreedCooldown()` updates cooldown with validation (1-168 hours)
- [ ] ADMIN-03: `updateRarityWeights()` validates weights sum to 10000
- [ ] ADMIN-04: `addNewSpecies()` adds species with unique ID and positive weight
- [ ] All admin endpoints require authentication and admin role
- [ ] All admin endpoints return 403 for non-admin users
- [ ] GET `/api/v2/game-config` returns current configuration
- [ ] Frontend admin page displays all config values with update forms
- [ ] Transaction hashes returned for all on-chain updates
- [ ] 12-block confirmation wait on all admin transactions

## Dependencies

- Admin role on users collection
- AdminConfig.sol contract deployed
- CONFIG_ADDRESS environment variable
- Platform wallet with admin private key for contract ownership
- Existing admin page structure at `/admin/`
