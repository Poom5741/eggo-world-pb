```typescript
test.describe("Marketplace Multi-User Journey", () => {
  test.describe.configure({ mode: "serial" })

  let listingId: string
  let tokenId: number
  let sellerBalanceBefore: number
  let buyerBalanceBefore: number

  // Pre-test: Verify seller has Animal NFT to list
  test("setup - seller has Animal NFT in inventory", async ({ page }) => {
    // Login as seller
    await e2eLogin(page, "test_seller", "/animals")

    // Verify seller has at least one Animal NFT
    const animalCards = page.locator("[data-animal-id], .animal-card")
    const count = await animalCards.count()
    expect(count).toBeGreaterThan(0)

    // Capture seller's on-chain Animal NFT balance before listing
    sellerBalanceBefore = await getBalanceOf(
      ANIMAL_NFT_ADDRESS,
      TEST_USERS.test_seller.walletAddress
    )

    // Store first animal's token_id for listing
    const firstCard = animalCards.first()
    const animalIdText = await firstCard.locator("[data-animal-id]").textContent()
    tokenId = parseInt(animalIdText?.match(/\d+/)?.[0] || "0")
    expect(tokenId).toBeGreaterThan(0)
  })

  // Step 1: Seller lists Animal NFT
  test("seller lists Animal NFT on marketplace", async ({ page }) => {
    await e2eLogin(page, "test_seller", "/animals")

    // Find animal card and click List button
    const animalCard = page.locator(`[data-animal-id="${tokenId}"]`).first()
    await animalCard.click()

    // Open list dialog
    const listButton = page.locator('button:has-text("List")')
    await expect(listButton).toBeVisible({ timeout: 5000 })
    await listButton.click()

    // Set price in dialog
    const priceInput = page.locator('input[type="number"]')
    await priceInput.fill("10") // 10 USDT

    // Confirm listing
    const confirmButton = page.locator('button:has-text("Confirm")')
    await confirmButton.click()

    // Wait for success toast
    await expect(page.locator('[data-sonner-toast][data-type="success"]')).toBeVisible({
      timeout: 10000,
    })

    // Verify listing created in marketplace_listings
    const { pocketbaseUrl } = getE2EContext()
    const response = await fetch(
      `${pocketbaseUrl}/api/collections/marketplace_listings/records?filter=(seller='${sellerUserId}')&filter=(nft_type='Animal')&filter=(status='active')`
    )
    const data = await response.json()
    expect(data.items.length).toBeGreaterThan(0)
    listingId = data.items[0].id
  })

  // Step 2: Buyer sees listing in marketplace
  test("buyer sees seller listing in marketplace", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/marketplace")

    // Capture buyer's on-chain balance before purchase
    buyerBalanceBefore = await getBalanceOf(ANIMAL_NFT_ADDRESS, TEST_USERS.test_buyer.walletAddress)

    // Find listing by seller or token_id
    const listingCard = page
      .locator(`[data-seller="${TEST_USERS.test_seller.walletAddress}"], .listing-card`)
      .first()
    await expect(listingCard).toBeVisible({ timeout: 10000 })

    // Verify price matches (10 USDT)
    const priceText = await listingCard.locator(".price").textContent()
    expect(priceText).toContain("10")
  })

  // Step 3: Buyer purchases listing
  test("buyer purchases listing and ownership transfers", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/marketplace")

    // Navigate to listing detail
    const listingCard = page.locator(".bg-surface-container-low.p-5.rounded-xl.clay-card").first()
    await listingCard.click()
    await page.waitForURL(/marketplace\/detail/, { timeout: 15000 })

    // Purchase flow
    const buyButton = page.locator('button:has-text("Buy")')
    await expect(buyButton).toBeVisible({ timeout: 5000 })
    await buyButton.click()

    const confirmDialog = page.locator('h2:has-text("Confirm Purchase")')
    await expect(confirmDialog).toBeVisible({ timeout: 5000 })

    const confirmButton = page.getByRole("button", { name: "Confirm Purchase" })
    await confirmButton.click()

    // Wait for redirect to animals/inventory
    await waitForPurchaseComplete(page, 30000)

    // Extract token ID from purchase
    const purchasedTokenId = await extractTokenIdFromPage(page)
    expect(purchasedTokenId).toBe(tokenId)
  })

  // Step 4: Verify ownership transfer
  test("ownership transfer verified - seller lost, buyer gained", async () => {
    const result = await verifyOwnershipTransfer(
      null, // page not needed for on-chain/PB checks
      tokenId,
      TEST_USERS.test_seller.walletAddress,
      TEST_USERS.test_buyer.walletAddress,
      "seller_user_id", // from PocketBase
      "buyer_user_id", // from PocketBase
      ANIMAL_NFT_ADDRESS
    )

    expect(result.transferComplete).toBe(true)
    expect(result.seller.hasOwnershipAfter).toBe(false)
    expect(result.buyer.hasOwnershipAfter).toBe(true)
    expect(result.buyer.onChainOwnerAfter.toLowerCase()).toBe(
      TEST_USERS.test_buyer.walletAddress.toLowerCase()
    )
  })

  // Step 5: Verify seller's inventory updated
  test("seller inventory no longer shows NFT", async ({ page }) => {
    await e2eLogin(page, "test_seller", "/animals")

    // Verify animal no longer in seller's inventory
    const animalCard = page.locator(`[data-animal-id="${tokenId}"]`)
    const visible = await animalCard.isVisible({ timeout: 5000 }).catch(() => false)
    expect(visible).toBe(false)

    // Verify on-chain balance decreased
    const sellerBalanceAfter = await getBalanceOf(
      ANIMAL_NFT_ADDRESS,
      TEST_USERS.test_seller.walletAddress
    )
    expect(sellerBalanceAfter).toBeLessThan(sellerBalanceBefore)
  })

  // Step 6: Verify buyer's inventory updated
  test("buyer inventory shows purchased NFT", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/animals")

    // Verify animal in buyer's inventory
    const animalCard = page.locator(`[data-animal-id="${tokenId}"]`)
    await expect(animalCard).toBeVisible({ timeout: 10000 })

    // Verify on-chain balance increased
    const buyerBalanceAfter = await getBalanceOf(
      ANIMAL_NFT_ADDRESS,
      TEST_USERS.test_buyer.walletAddress
    )
    expect(buyerBalanceAfter).toBeGreaterThan(buyerBalanceBefore)
  })
})
```
