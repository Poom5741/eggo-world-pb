**Commands to execute:**

```bash
cd /Users/poom-work/tokenine/eggo-pocketbase

# Read addresses from the extracted data
set USDT (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['usdt'])")
set COMMISSION (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['commission'])")
set ANIMAL_NFT (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['animalNft'])")
set EGG_NFT (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['eggNft'])")
set FOOD_NFT (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['foodNft'])")
set MARKETPLACE (python3 -c "import json; d=json.load(open('contracts/contract-addresses.json')); print(d['7117']['marketplace'])")

echo "USDT: $USDT"
echo "Commission: $COMMISSION"
echo "AnimalNFT: $ANIMAL_NFT"
echo "EggNFT: $EGG_NFT"
echo "FoodNFT: $FOOD_NFT"
echo "Marketplace: $MARKETPLACE"

# Update wallet-api/.env contract addresses
python3 -c "
import re

# Read current .env
with open('wallet-api/.env', 'r') as f:
    content = f.read()

# Define new addresses
addresses = {
    'MARKETPLACE_CONTRACT_ADDRESS': '$MARKETPLACE',
    'EGG_NFT_CONTRACT_ADDRESS': '$EGG_NFT',
    'FOOD_NFT_CONTRACT_ADDRESS': '$FOOD_NFT',
    'ANIMAL_NFT_CONTRACT_ADDRESS': '$ANIMAL_NFT',
}

# Update or append each address
for key, value in addresses.items():
    pattern = re.compile(r'^' + re.escape(key) + r'=.*', re.MULTILINE)
    if pattern.search(content):
        content = pattern.sub(f'{key}={value}', content)
        print(f'Updated {key}')
    else:
        content += f'\n{key}={value}\n'
        print(f'Appended {key}')

# Write back
with open('wallet-api/.env', 'w') as f:
    f.write(content)

print('wallet-api/.env updated successfully')
"
```
