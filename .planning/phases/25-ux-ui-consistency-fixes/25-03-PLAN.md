### Step 2: Create Shadow/Border Migration Script

**File:** `scripts/migrate-shadow-radius.sh` (temporary migration tool)

```bash
#!/bin/bash
# Migrate shadow and border radius to claymorphism standards

# Find all TSX files in the app directory
FILES=$(find apps/web -name "*.tsx" -o -name "*.ts" | grep -v node_modules | grep -v ".next")

# Shadow migration (shadow-lg → shadow-clay-lg, shadow-md → shadow-clay-md)
for file in $FILES; do
  if grep -q "shadow-lg\|shadow-md" "$file"; then
    sed -i '' 's/shadow-lg/shadow-clay-lg/g' "$file"
    sed -i '' 's/shadow-md/shadow-clay-md/g' "$file"
  fi
done

# Border radius migration (rounded-xl → rounded-clay-lg, rounded-lg → rounded-clay)
for file in $FILES; do
  if grep -q "rounded-xl\|rounded-lg" "$file"; then
    sed -i '' 's/rounded-xl/rounded-clay-lg/g' "$file"
    sed -i '' 's/rounded-lg/rounded-clay/g' "$file"
  fi
done

echo "Migration complete. Please verify visually."
```

## Plan 25-03-02: Layout Standardization

### Step 1: Create Unified Layout Component

**File:** `apps/web/components/LayoutStandard.tsx` (new)

```tsx
"use client"

import { LayoutWrapper } from "./LayoutWrapper"
import { LayoutWithoutNav } from "./LayoutWithoutNav"
import Header from "./Header"

interface LayoutStandardProps {
  children: React.ReactNode
  variant?: "full" | "with-header" | "minimal"
}

export function LayoutStandard({ children, variant = "full" }: LayoutStandardProps) {
  if (variant === "minimal") {
    // No navigation (auth pages, modals)
    return <>{children}</>
  }

  if (variant === "with-header") {
    // Only header, no side/bottom nav
    return (
      <LayoutWithoutNav>
        <Header />
        {children}
      </LayoutWithoutNav>
    )
  }

  // Full navigation (default) - LayoutWrapper
  return <LayoutWrapper>{children}</LayoutWrapper>
}

export default LayoutStandard
```

## Plan 25-03-03: Interaction Polish (cursor-pointer + transitions)

### Step 1: Audit Current Usage

**Find all clickable elements:**

```bash
# Find onClick handlers without cursor-pointer
rg "onClick=" --glob "*.tsx" -l | while read file; do
  if grep -q "onClick" "$file" && ! grep -q "cursor-pointer" "$file"; then
    echo "$file (has onClick but no cursor-pointer)"
  fi
done

# Find hover states without transitions
rg "hover:" --glob "*.tsx" -l | while read file; do
  if grep -q "hover:" "$file" && ! grep -q "transition-" "$file"; then
    echo "$file (has hover but no transition)"
  fi
done
```

**Expected counts from audit:**

- `cursor-pointer` missing: 92 instances across 58 files
- Transitions missing: 42 files with hover states
