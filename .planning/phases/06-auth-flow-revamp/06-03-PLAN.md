From apps/web/lib/pocketbase/client.ts (createClient is singleton, safe to call synchronously):

```typescript
export function createClient(): PocketBase {
  if (!pb) {
    pb = new PocketBase(POCKETBASE_URL)
    // Loads auth from localStorage if available
    if (typeof window !== "undefined") {
      const stored = localStorage.getItem("pocketbase_auth")
      if (stored) {
        const { token, model } = JSON.parse(stored)
        pb.authStore.save(token, model)
      }
      // ...
    }
  }
  return pb // returns singleton — safe to call multiple times
}
```
