Test structure (per D-45-03: serial mode for dependent tests):

```typescript
test.describe("Feed + Hatch Journey", () => {
  test.describe.configure({ mode: "serial" })

  let foodIds: number[] = []
  let eggTokenId: number
  let animalTokenId: number

  // Step 1: Login and get ready state
  test("setup - user has egg ready for feeding", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/eggs")

    // Verify user has egg with food_count < 10
    const eggCards = page.locator("[data-egg-id], .egg-card")
    const count = await eggCards.count()
    expect(count).toBeGreaterThan(0)

    // Get egg tokenId from page or PocketBase
    eggTokenId = await getEggTokenIdForUser(userId)
    expect(eggTokenId).toBeGreaterThan(0)
  })

  // Step 2: Buy food from marketplace
  test("buy food from marketplace", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/marketplace")

    // Find food listing and purchase
    foodIds = await buyFoodFromMarketplace(page, 10)
    expect(foodIds.length).toBeGreaterThanOrEqual(10)
  })

  // Step 3: Feed the egg
  test("feed egg with purchased food", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/eggs")

    // Click on egg card to open feed dialog
    const eggCard = page.locator(`[data-egg-id="${eggTokenId}"]`).first()
    await eggCard.click()

    // Click "Manage" or "Feed" button
    const feedButton = page.locator('button:has-text("Feed"), button:has-text("Manage")')
    await expect(feedButton).toBeVisible({ timeout: 5000 })
    await feedButton.click()

    // In FeedDialog, select all 10 food items
    const foodCards = page.locator('[role="listitem"]')
    const foodCount = await foodCards.count()
    expect(foodCount).toBeGreaterThanOrEqual(10)

    // Select 10 items by clicking them
    for (let i = 0; i < 10; i++) {
      await foodCards.nth(i).click()
    }

    // Click "Feed 10 items" button
    const confirmFeedButton = page.locator('button:has-text("Feed 10")')
    await confirmFeedButton.click()

    // Wait for feed to complete
    await waitForFeedComplete(page, 15000)

    // Verify egg now shows 10/10 progress
    await page.goto("/eggs/")
    const progressText = await page.locator(`[data-egg-id="${eggTokenId}"] .progress`).textContent()
    expect(progressText).toContain("10/10")
  })

  // Step 4: Hatch the egg
  test("hatch egg and verify animal appears", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/eggs")

    // Find egg with 10/10 and click Hatch button
    const eggCard = page.locator(`[data-egg-id="${eggTokenId}"]`).first()
    await eggCard.click()

    // Click Hatch button
    const hatchButton = page.locator('button:has-text("Hatch")')
    await expect(hatchButton).toBeVisible({ timeout: 5000 })
    await hatchButton.click()

    // Wait for hatch modal to open
    const hatchModal = page.locator('h2:has-text("Ready to Hatch")')
    await expect(hatchModal).toBeVisible({ timeout: 5000 })

    // Click HATCH! button in modal
    const confirmHatchButton = page.locator('button:has-text("HATCH")')
    await confirmHatchButton.click()

    // Wait for hatch animation to complete
    animalTokenId = await waitForHatchComplete(page, 30000)
    expect(animalTokenId).toBeGreaterThan(0)

    // Click Continue to go to animals page
    const continueButton = page.locator('button:has-text("Continue")')
    await continueButton.click()
  })

  // Step 5: Triple verification
  test("verify animal ownership - triple verification", async ({ page }) => {
    await e2eLogin(page, "test_buyer", "/animals")

    const result = await verifyAnimalOwnership(
      page,
      animalTokenId,
      TEST_USERS.test_buyer.walletAddress,
      userId
    )

    expect(result.uiVisible).toBe(true)
    expect(result.onChainOwner.toLowerCase()).toBe(
      TEST_USERS.test_buyer.walletAddress.toLowerCase()
    )
    expect(result.allMatch).toBe(true)
  })

  // Error scenario: No food available
  test("no food available shows message - cannot feed", async ({ page }) => {
    await e2eLogin(page, "test_buyer_poor", "/eggs")

    // Try to feed - should see "No food available"
    const eggCard = page.locator(".egg-card").first()
    await eggCard.click()

    const feedButton = page.locator('button:has-text("Feed"), button:has-text("Manage")')
    await feedButton.click()

    // Verify "No food available" message
    const noFoodMessage = page.locator("text=No food available")
    await expect(noFoodMessage).toBeVisible({ timeout: 5000 })
  })
})
```
