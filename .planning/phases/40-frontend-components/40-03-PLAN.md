```typescript
interface BurnNFTDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  nftId?: string // pre-filled NFT ID (optional)
  nftType?: "egg" | "food" | "animal" // pre-filled NFT type (optional)
  onSuccess?: (txHash: string) => void // callback on success
}

export function BurnNFTDialog({
  open,
  onOpenChange,
  nftId,
  nftType,
  onSuccess,
}: BurnNFTDialogProps) {
  const isHydrated = useIsHydrated()
  const [isBurning, setIsBurning] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const handleBurn = async () => {
    if (!nftId || !nftType) return
    setIsBurning(true)
    setError(null)
    try {
      const token = pb.authStore.token
      const response = await fetch(`${pb.baseURL}/api/v2/burn-nft`, {
        method: "POST",
        headers: {
          Authorization: token,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ nft_id: nftId, nft_type: nftType }),
      })
      const data = await response.json()
      if (!data.success) throw new Error(data.error.message)
      onSuccess?.(data.data.transaction_hash)
      onOpenChange(false)
    } catch (err) {
      setError(err instanceof Error ? err.message : "Burn failed")
    } finally {
      setIsBurning(false)
    }
  }

  if (!isHydrated) return null
  // ... render dialog
}
```
