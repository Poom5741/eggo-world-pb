```javascript
// Register route for NFT purchase
routerAdd("POST", "/api/v2/marketplace/buy", async (e) => {
  const data = e.requestInfo()
  const { listing_id, buyer_address } = data

  if (!listing_id || !buyer_address) {
    return e.json(400, {
      success: false,
      error: { message: "Missing listing_id or buyer_address" },
    })
  }

  // Get authenticated user
  const authRecord = e.auth
  if (!authRecord) {
    return e.json(401, {
      success: false,
      error: { message: "Authentication required" },
    })
  }

  // Fetch listing from marketplace_listings collection
  const listing = $app.dao().findRecordById("marketplace_listings", listing_id)

  if (!listing || listing.get("status") !== "active") {
    return e.json(404, {
      success: false,
      error: { message: "Listing not found or not active" },
    })
  }

  const price = listing.get("price")
  const seller_id = listing.get("seller")
  const nft_type = listing.get("nft_type")
  const nft_id = listing.get("nft_id")

  // Get buyer and seller records
  const buyer = authRecord
  const seller = $app.dao().findRecordById("users", seller_id)

  // Check buyer has sufficient USDT balance
  const buyerBalance = buyer.get("usdt_balance") || 0
  if (buyerBalance < price) {
    return e.json(400, {
      success: false,
      error: {
        message: `Insufficient balance. Required: ${price} USDT, Available: ${buyerBalance} USDT`,
        code: "INSUFFICIENT_BALANCE",
      },
    })
  }

  // Calculate fees (4% marketplace fee per BuyFlow component comment)
  const fee = Math.floor(price * 0.04)
  const sellerAmount = price - fee

  // Execute atomic transfer in transaction
  const collection = $app.dao().findCollectionByNameOrId("users")

  // Deduct from buyer
  buyer.set("usdt_balance", buyerBalance - price)
  $app.dao().saveRecord(buyer)

  // Credit seller (minus fee)
  const sellerBalance = seller.get("usdt_balance") || 0
  seller.set("usdt_balance", sellerBalance + sellerAmount)
  $app.dao().saveRecord(seller)

  // Transfer NFT ownership (update nft record owner to buyer)
  let nftCollection
  if (nft_type === "egg") {
    nftCollection = $app.dao().findCollectionByNameOrId("egg_nfts")
  } else if (nft_type === "food") {
    nftCollection = $app.dao().findCollectionByNameOrId("food_nfts")
  } else if (nft_type === "animal") {
    nftCollection = $app.dao().findCollectionByNameOrId("animal_nfts")
  }

  if (nftCollection) {
    const nft = $app.dao().findRecordById(nftCollection.name, nft_id)
    nft.set("owner", buyer.id)
    $app.dao().saveRecord(nft)
  }

  // Mark listing as sold
  listing.set("status", "sold")
  listing.set("buyer", buyer.id)
  $app.dao().saveRecord(listing)

  // Log transaction
  const txCollection = $app.dao().findCollectionByNameOrId("transactions")
  const txRecord = new Record(txCollection)
  txRecord.set("user", buyer.id)
  txRecord.set("type", "nft_purchase")
  txRecord.set("amount", price)
  txRecord.set("nft_type", nft_type)
  txRecord.set("nft_id", nft_id)
  txRecord.set("status", "completed")
  $app.dao().saveRecord(txRecord)

  return e.json(200, {
    success: true,
    data: {
      listing_id: listing_id,
      buyer_address: buyer_address,
      price: price,
      fee: fee,
      seller_amount: sellerAmount,
      nft_type: nft_type,
      nft_id: nft_id,
      tx_hash: txRecord.id,
    },
  })
})
```
